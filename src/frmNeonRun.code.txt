Option Explicit
' Окно перед расчётом (если включено «Спрашивать при каждом запуске»): диаметр фрезы,
' тип неона, кратность реза, подгонка. Элементы окна создаёт NeonBuild, код вставляется при сборке.
' NeonRunMode = 1 — то же окно перед упрощением контура: допуск и «показывать окно» (лишние
' элементы скрываются, окно становится короче — сборщик менять не нужно).
' NeonRunMode = 2 — перед расстановкой люверсов: отступ от края, расстояние между люверсами,
' диаметр и цвет точки, усиление углов (элементы те же, подписи и места — свои).
' NeonRunMode = 3 — перед раскладкой на листы: размер листа, поле, зазор, по цвету, ширина по раскладке.

' люверсы: свой цвет ("S:..."), последний выбранный цвет из списка, защита от повторного Change
Private gCustom As String, gLastIdx As Long, gBusy As Boolean

Private Sub UserForm_Initialize()
    If NeonRunMode = 1 Then
        InitSimplify
        Exit Sub
    End If
    If NeonRunMode = 2 Then
        InitGrommets
        Exit Sub
    End If
    If NeonRunMode = 3 Then
        InitNest
        Exit Sub
    End If
    Me.Caption = NeonL("app_title")
    lblTool.Caption = NeonL("lbl_tool")
    cboTool.Clear
    cboTool.AddItem "6"
    cboTool.AddItem "8"
    cboTool.AddItem "10"
    cboTool.Text = FmtNum(NeonGetTool())
    lblHint.Caption = NeonL("lbl_tool_hint")
    lblEnd.Caption = NeonL("lbl_endoff")
    cboEnd.Clear
    cboEnd.AddItem NeonL("endoff_default")
    cboEnd.AddItem "0"
    cboEnd.AddItem "2"
    cboEnd.AddItem "5"
    If Len(GetSetting("NeonCenterline", "Params", "EndOff", "")) = 0 Then
        cboEnd.ListIndex = 0
    Else
        cboEnd.Text = GetSetting("NeonCenterline", "Params", "EndOff", "")
    End If
    lblType.Caption = NeonL("lbl_type")
    cboType.Clear
    cboType.AddItem NeonL("type_classic")
    cboType.AddItem NeonL("type_neon2")
    cboType.ListIndex = IIf(NeonCutType() = "neon2", 1, 0)
    lblRef.Caption = NeonL("lbl_ref")
    cboRef.Clear
    cboRef.AddItem NeonL("ref_off")
    cboRef.AddItem NeonL("ref_tip")
    cboRef.AddItem NeonL("ref_half")
    cboRef.ListIndex = CLng(Val(GetSetting("NeonCenterline", "Script", "RefEnd", "1")))
    lblCut.Caption = NeonL("lbl_cut")
    cboCut.Clear
    cboCut.AddItem "10"
    cboCut.AddItem "25"
    cboCut.AddItem "50"
    cboCut.Text = FmtNum(NeonCutStep())
    chkFit.Caption = NeonL("chk_fit")
    chkFit.Value = NeonCutFitOn()
    lblTrim.Caption = NeonL("lbl_trim")
    cboTrim.Clear
    cboTrim.AddItem NeonL("trim_both")
    cboTrim.AddItem NeonL("trim_end")
    cboTrim.ListIndex = IIf(GetSetting("NeonCenterline", "Cut", "Trim", "both") = "end", 1, 0)
    chkAsk.Caption = NeonL("chk_ask_run")
    chkAsk.Value = True
    btnOK.Caption = NeonL("btn_run")
    btnCancel.Caption = NeonL("btn_cancel")
    NeonRunConfirmed = False
End Sub

Private Sub InitSimplify()
    Dim c As Object
    For Each c In Me.Controls
        c.Visible = False
    Next c
    Me.Caption = NeonL("simp_title")
    lblTool.Caption = NeonL("lbl_simp_tol")
    lblTool.Visible = True
    cboTool.Clear
    cboTool.AddItem "0.05"
    cboTool.AddItem "0.1"
    cboTool.AddItem "0.25"
    cboTool.Text = FmtNum(NeonSimplifyTol())
    cboTool.Visible = True
    lblHint.WordWrap = True
    lblHint.Height = 42
    lblHint.Caption = NeonL("simp_hint")
    lblHint.Visible = True
    chkAsk.Top = lblHint.Top + lblHint.Height + 8
    chkAsk.Caption = NeonL("chk_simp_ask")
    chkAsk.Value = True
    chkAsk.Visible = True
    btnOK.Top = chkAsk.Top + chkAsk.Height + 12
    btnCancel.Top = btnOK.Top
    btnOK.Caption = NeonL("btn_simp")
    btnCancel.Caption = NeonL("btn_cancel")
    btnOK.Visible = True
    btnCancel.Visible = True
    Me.Height = btnOK.Top + btnOK.Height + 50
    NeonRunConfirmed = False
End Sub

Private Sub InitGrommets()
    Dim c As Object, cs As String, i As Long, v As Variant, rd As Double
    gBusy = True
    For Each c In Me.Controls
        c.Visible = False
    Next c
    Me.Caption = NeonL("grom_title")
    GromRow lblTool, cboTool, 14, NeonL("lbl_grom_edge")
    cboTool.Clear
    For Each v In Array("10", "15", "20", "25", "30")
        cboTool.AddItem v
    Next v
    cboTool.Text = FmtNum(NeonGromEdge())
    GromRow lblEnd, cboEnd, 40, NeonL("lbl_grom_step")
    cboEnd.Clear
    For Each v In Array("200", "250", "300", "400", "500")
        cboEnd.AddItem v
    Next v
    cboEnd.Text = FmtNum(NeonGromStep())
    GromRow lblCut, cboCut, 66, NeonL("lbl_grom_dot")
    cboCut.Clear
    For Each v In Array("3", "5", "8", "10")
        cboCut.AddItem v
    Next v
    cboCut.Text = FmtNum(NeonGromDot())
    lblType.Left = 12: lblType.Top = 96: lblType.Width = 120
    lblType.Caption = NeonL("lbl_grom_color")
    lblType.Visible = True
    cboType.Left = 140: cboType.Top = 92: cboType.Width = 198
    cboType.Clear
    For Each v In Array("grom_c_black", "grom_c_red", "grom_c_magenta", "grom_c_blue", "grom_c_green", "grom_c_white", "grom_c_other")
        cboType.AddItem NeonL(CStr(v))
    Next v
    cboType.Visible = True
    cs = NeonGromColorSetting()
    gCustom = "": gLastIdx = 0
    If Left$(cs, 2) = "S:" Then
        gCustom = cs
        cboType.List(cboType.ListCount - 1) = NeonL("grom_c_other") & ": " & NeonGromColorName(cs)
        cboType.ListIndex = cboType.ListCount - 1
    Else
        i = CLng(Val(Mid$(cs, 3)))
        If i < 0 Or i > cboType.ListCount - 2 Then
            i = 0
        End If
        cboType.ListIndex = i
        gLastIdx = i
    End If
    chkFit.Left = 12: chkFit.Top = 122: chkFit.Width = 230
    chkFit.Caption = NeonL("chk_grom_reinf")
    chkFit.Value = (GetSetting("NeonCenterline", "Grommet", "Reinf", "0") = "1")
    chkFit.Visible = True
    cboTrim.Left = 250: cboTrim.Top = 120: cboTrim.Width = 88
    cboTrim.Clear
    For Each v In Array("30", "40", "50", "60", "75", "100")
        cboTrim.AddItem v
    Next v
    rd = NeonGromReinfDist()
    cboTrim.ListIndex = 2
    For i = 0 To cboTrim.ListCount - 1
        If Abs(Val(cboTrim.List(i)) - rd) < 0.001 Then
            cboTrim.ListIndex = i
        End If
    Next i
    cboTrim.Visible = True
    lblHint.Left = 12: lblHint.Top = 150: lblHint.Width = 326
    lblHint.WordWrap = True
    lblHint.Height = 44
    lblHint.Caption = NeonL("grom_hint")
    lblHint.Visible = True
    btnOK.Top = lblHint.Top + lblHint.Height + 10
    btnCancel.Top = btnOK.Top
    btnOK.Caption = NeonL("btn_grom")
    btnCancel.Caption = NeonL("btn_cancel")
    btnOK.Visible = True
    btnCancel.Visible = True
    Me.Height = btnOK.Top + btnOK.Height + 50
    NeonRunConfirmed = False
    gBusy = False
End Sub

Private Sub InitNest()
    Dim c As Object, v As Variant, wm As Double, i As Long
    For Each c In Me.Controls
        c.Visible = False
    Next c
    Me.Caption = NeonL("nest_title")
    GromRow lblTool, cboTool, 14, NeonL("lbl_nest_size")
    ' список: свои размеры (последние введённые), затем стандартные листы и стол лазера
    cboTool.Clear
    For Each v In Split(GetSetting("NeonCenterline", "Nest", "Recent", "") & ";3050x2050;3050x2030;3000x2000;2050x1525;" & _
                        "2000x1000;1525x1025;1000x600;900x600;600x400;500x300", ";")
        If Len(v) > 0 And Not NestInList(CStr(v)) Then
            cboTool.AddItem v
        End If
    Next v
    cboTool.Text = NeonNestSize()
    GromRow lblEnd, cboEnd, 40, NeonL("lbl_nest_margin")
    cboEnd.Clear
    For Each v In Array("0", "5", "10", "15", "20")
        cboEnd.AddItem v
    Next v
    cboEnd.Text = FmtNum(NeonNestNum("Margin", 10))
    GromRow lblCut, cboCut, 66, NeonL("lbl_nest_gap")
    cboCut.Clear
    For Each v In Array("3", "5", "8", "10", "12", "15")
        cboCut.AddItem v
    Next v
    cboCut.Text = FmtNum(NeonNestNum("Gap", 10))
    chkFit.Left = 12: chkFit.Top = 94: chkFit.Width = 326
    chkFit.Caption = NeonL("chk_nest_color")
    chkFit.Value = NeonNestFlag("ByColor", True)
    chkFit.Visible = True
    chkAsk.Left = 12: chkAsk.Top = 120: chkAsk.Width = 232
    chkAsk.Caption = NeonL("chk_nest_auto")
    chkAsk.Value = NeonNestFlag("AutoW", False)
    chkAsk.Visible = True
    ' наименьшая ширина: список с правом ввести своё (если окно позволяет)
    On Error Resume Next
    cboTrim.Style = 0
    On Error GoTo 0
    cboTrim.Left = 250: cboTrim.Top = 118: cboTrim.Width = 88
    cboTrim.Clear
    For Each v In Array("0", "300", "500", "1000", "1500", "2000")
        cboTrim.AddItem v
    Next v
    wm = NeonNestNum("WMin", 0)
    cboTrim.ListIndex = 0
    For i = 0 To cboTrim.ListCount - 1
        If Abs(Val(cboTrim.List(i)) - wm) < 0.001 Then
            cboTrim.ListIndex = i
        End If
    Next i
    On Error Resume Next
    cboTrim.Text = FmtNum(wm)
    On Error GoTo 0
    cboTrim.Enabled = chkAsk.Value
    cboTrim.Visible = True
    lblHint.Left = 12: lblHint.Top = 148: lblHint.Width = 326
    lblHint.WordWrap = True
    lblHint.Height = 108
    lblHint.Caption = NeonL("nest_hint")
    lblHint.Visible = True
    btnOK.Top = lblHint.Top + lblHint.Height + 10
    btnCancel.Top = btnOK.Top
    btnOK.Caption = NeonL("btn_nest")
    btnCancel.Caption = NeonL("btn_cancel")
    btnOK.Visible = True
    btnCancel.Visible = True
    Me.Height = btnOK.Top + btnOK.Height + 50
    NeonRunConfirmed = False
End Sub

' раскладка: размер уже есть в списке листов
Private Function NestInList(ByVal s As String) As Boolean
    Dim i As Long
    For i = 0 To cboTool.ListCount - 1
        If cboTool.List(i) = s Then
            NestInList = True
            Exit Function
        End If
    Next i
End Function

' раскладка: наименьшая ширина — только с галочкой «Ширина по раскладке»
Private Sub chkAsk_Click()
    If NeonRunMode = 3 Then
        cboTrim.Enabled = chkAsk.Value
    End If
End Sub

' строка окна люверсов: подпись слева, поле справа
Private Sub GromRow(lb As Object, cb As Object, ByVal y As Single, ByVal cap As String)
    lb.Left = 12: lb.Top = y + 4: lb.Width = 232
    lb.Caption = cap
    lb.Visible = True
    cb.Left = 250: cb.Top = y: cb.Width = 88
    cb.Visible = True
End Sub

' люверсы: «Другой…» в списке цветов — окно выбора цвета CorelDRAW
Private Sub cboType_Change()
    Dim s As String, cur As String
    If NeonRunMode <> 2 Or gBusy Then
        Exit Sub
    End If
    If cboType.ListIndex <> cboType.ListCount - 1 Then
        gLastIdx = cboType.ListIndex
        Exit Sub
    End If
    gBusy = True
    cur = gCustom
    If Len(cur) = 0 Then
        cur = "P:" & gLastIdx
    End If
    s = NeonGromPickColor(cur)
    If Len(s) > 0 Then
        gCustom = s
        cboType.List(cboType.ListCount - 1) = NeonL("grom_c_other") & ": " & NeonGromColorName(s)
    ElseIf Len(gCustom) = 0 Then
        cboType.ListIndex = gLastIdx
    End If
    gBusy = False
End Sub

Private Sub btnOK_Click()
    Dim t As Double, c As Double, st As Double, d As Double, w As Double, h As Double
    If NeonRunMode = 3 Then
        t = ParseNum(cboEnd.Text): c = ParseNum(cboCut.Text): d = ParseNum(cboTrim.Text)
        If Not NeonNestParse(cboTool.Text, w, h) Then
            MsgBox NeonL("bad_nest"), vbExclamation, NeonL("nest_title")
            Exit Sub
        End If
        If t < 0 Or t > 1000 Or c < 0 Or c > 1000 Or d < 0 Or 2 * t >= w Or 2 * t >= h Or (chkAsk.Value And d > w) Then
            MsgBox NeonL("bad_nest"), vbExclamation, NeonL("nest_title")
            Exit Sub
        End If
        SaveSetting "NeonCenterline", "Nest", "Size", FmtNum(w) & "x" & FmtNum(h)
        NestRemember FmtNum(w) & "x" & FmtNum(h)
        SaveSetting "NeonCenterline", "Nest", "Margin", FmtNum(t)
        SaveSetting "NeonCenterline", "Nest", "Gap", FmtNum(c)
        SaveSetting "NeonCenterline", "Nest", "ByColor", IIf(chkFit.Value, "1", "0")
        SaveSetting "NeonCenterline", "Nest", "AutoW", IIf(chkAsk.Value, "1", "0")
        SaveSetting "NeonCenterline", "Nest", "WMin", FmtNum(d)
        NeonRunConfirmed = True
        Me.Hide
        Exit Sub
    End If
    If NeonRunMode = 2 Then
        t = ParseNum(cboTool.Text): st = ParseNum(cboEnd.Text): d = ParseNum(cboCut.Text)
        If t <= 0 Or t > 500 Or st < 20 Or st > 10000 Or d <= 0 Or d > 100 Then
            MsgBox NeonL("bad_grom"), vbExclamation, NeonL("grom_title")
            Exit Sub
        End If
        SaveSetting "NeonCenterline", "Grommet", "Edge", FmtNum(t)
        SaveSetting "NeonCenterline", "Grommet", "Step", FmtNum(st)
        SaveSetting "NeonCenterline", "Grommet", "Dot", FmtNum(d)
        If cboType.ListIndex = cboType.ListCount - 1 Then
            If Len(gCustom) > 0 Then
                SaveSetting "NeonCenterline", "Grommet", "Color", gCustom
            End If
        ElseIf cboType.ListIndex >= 0 Then
            SaveSetting "NeonCenterline", "Grommet", "Color", "P:" & cboType.ListIndex
        End If
        SaveSetting "NeonCenterline", "Grommet", "Reinf", IIf(chkFit.Value, "1", "0")
        If cboTrim.ListIndex >= 0 Then
            SaveSetting "NeonCenterline", "Grommet", "ReinfDist", cboTrim.List(cboTrim.ListIndex)
        End If
        NeonRunConfirmed = True
        Me.Hide
        Exit Sub
    End If
    If NeonRunMode = 1 Then
        t = ParseNum(cboTool.Text)
        If t < 0.01 Or t > 5 Then
            MsgBox NeonL("bad_simp"), vbExclamation, NeonL("simp_title")
            Exit Sub
        End If
        SaveSetting "NeonCenterline", "Simplify", "Tol", FmtNum(t)
        SaveSetting "NeonCenterline", "Simplify", "Ask", IIf(chkAsk.Value, "1", "0")
        NeonRunConfirmed = True
        Me.Hide
        Exit Sub
    End If
    t = ParseNum(cboTool.Text)
    If t <= 0 Or t > 100 Then
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("app_title")
        Exit Sub
    End If
    c = ParseNum(cboCut.Text)
    If c < 0 Or c > 1000 Then
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("app_title")
        Exit Sub
    End If
    If cboEnd.ListIndex = 0 Or Len(Trim$(cboEnd.Text)) = 0 Then
        SaveSetting "NeonCenterline", "Params", "EndOff", ""
    ElseIf ParseNum(cboEnd.Text) >= 0 And ParseNum(cboEnd.Text) <= 100 Then
        SaveSetting "NeonCenterline", "Params", "EndOff", FmtNum(ParseNum(cboEnd.Text))
    Else
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("app_title")
        Exit Sub
    End If
    SaveSetting "NeonCenterline", "Params", "Tool", FmtNum(t)
    SaveSetting "NeonCenterline", "Cut", "Step", FmtNum(c)
    SaveSetting "NeonCenterline", "Script", "RefEnd", CStr(cboRef.ListIndex)
    SaveSetting "NeonCenterline", "Cut", "Type", IIf(cboType.ListIndex = 1, "neon2", "classic")
    SaveSetting "NeonCenterline", "Cut", "Fit", IIf(chkFit.Value, "1", "0")
    SaveSetting "NeonCenterline", "Cut", "Trim", IIf(cboTrim.ListIndex = 1, "end", "both")
    SaveSetting "NeonCenterline", "UI", "AskTool", IIf(chkAsk.Value, "1", "0")
    NeonRunConfirmed = True
    Me.Hide
End Sub

' раскладка: размер листа — первым в список последних (не больше шести)
Private Sub NestRemember(ByVal s As String)
    Dim a() As String, r As String, i As Long, n As Long
    r = s: n = 1
    a = Split(GetSetting("NeonCenterline", "Nest", "Recent", ""), ";")
    For i = 0 To UBound(a)
        If Len(a(i)) > 0 And a(i) <> s And n < 6 Then
            r = r & ";" & a(i)
            n = n + 1
        End If
    Next i
    SaveSetting "NeonCenterline", "Nest", "Recent", r
End Sub

Private Sub btnCancel_Click()
    NeonRunConfirmed = False
    Me.Hide
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    NeonRunConfirmed = False
End Sub
