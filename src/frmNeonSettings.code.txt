Option Explicit
' Окно настроек NeonCenterline. Элементы окна создаёт NeonBuild (сборщик), этот код
' вставляется в форму при сборке. Строка «Упрощение контура» добавляется при открытии окна
' (AddSimpRow) — сборщик менять не нужно.

Private lblSimp As Object, cboSimp As Object

Private Sub UserForm_Initialize()
    Dim lg As String
    Me.Caption = NeonL("app_title") & " - " & NeonL("settings_title")
    AddSimpRow
    If Not cboSimp Is Nothing Then
        lblSimp.Caption = NeonL("lbl_simp")
        cboSimp.Clear
        cboSimp.AddItem "0.05"
        cboSimp.AddItem "0.1"
        cboSimp.AddItem "0.25"
        cboSimp.Text = FmtNum(NeonSimplifyTol())
    End If
    lblTool.Caption = NeonL("lbl_tool")
    cboTool.Clear
    cboTool.AddItem "6"
    cboTool.AddItem "8"
    cboTool.AddItem "10"
    cboTool.Text = FmtNum(NeonGetTool())
    lblHint.Caption = NeonL("lbl_tool_hint")
    lblEnd.Caption = NeonL("lbl_endoff")
    cboEnd.Clear
    cboEnd.AddItem NeonL("endoff_default")
    cboEnd.AddItem "0"
    cboEnd.AddItem "2"
    cboEnd.AddItem "5"
    If Len(GetSetting("NeonCenterline", "Params", "EndOff", "")) = 0 Then
        cboEnd.ListIndex = 0
    Else
        cboEnd.Text = GetSetting("NeonCenterline", "Params", "EndOff", "")
    End If
    chkAsk.Caption = NeonL("chk_ask")
    chkAsk.Value = (GetSetting("NeonCenterline", "UI", "AskTool", "1") = "1")
    lblType.Caption = NeonL("lbl_type")
    cboType.Clear
    cboType.AddItem NeonL("type_classic")
    cboType.AddItem NeonL("type_neon2")
    cboType.ListIndex = IIf(NeonCutType() = "neon2", 1, 0)
    lblRef.Caption = NeonL("lbl_ref")
    cboRef.Clear
    cboRef.AddItem NeonL("ref_off")
    cboRef.AddItem NeonL("ref_tip")
    cboRef.AddItem NeonL("ref_half")
    cboRef.ListIndex = CLng(Val(GetSetting("NeonCenterline", "Script", "RefEnd", "1")))
    lblCut.Caption = NeonL("lbl_cut")
    cboCut.Clear
    cboCut.AddItem "10"
    cboCut.AddItem "25"
    cboCut.AddItem "50"
    cboCut.Text = FmtNum(NeonCutStep())
    chkFit.Caption = NeonL("chk_fit")
    chkFit.Value = NeonCutFitOn()
    lblTrim.Caption = NeonL("lbl_trim")
    cboTrim.Clear
    cboTrim.AddItem NeonL("trim_both")
    cboTrim.AddItem NeonL("trim_end")
    cboTrim.ListIndex = IIf(GetSetting("NeonCenterline", "Cut", "Trim", "both") = "end", 1, 0)
    lblLang.Caption = NeonL("lbl_lang")
    cboLang.Clear
    cboLang.AddItem NeonL("lang_auto")
    cboLang.AddItem NeonL("lang_ru")
    cboLang.AddItem NeonL("lang_en")
    lg = LCase$(GetSetting("NeonCenterline", "UI", "Lang", ""))
    If lg = "ru" Then
        cboLang.ListIndex = 1
    ElseIf lg = "en" Then
        cboLang.ListIndex = 2
    Else
        cboLang.ListIndex = 0
    End If
    chkUpd.Caption = NeonL("chk_upd")
    chkUpd.Value = (GetSetting("NeonCenterline", "Update", "Daily", "1") = "1")
    lblVer.Caption = NeonLF("lbl_version", NEON_VERSION)
    btnUpdate.Caption = NeonL("btn_update")
    btnAbout.Caption = NeonL("btn_about")
    btnOK.Caption = NeonL("btn_ok")
    btnCancel.Caption = NeonL("btn_cancel")
End Sub

Private Sub btnOK_Click()
    Dim t As Double, c As Double, lg As String, sm As Double
    t = ParseNum(cboTool.Text)
    If t <= 0 Or t > 100 Then
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("settings_title")
        Exit Sub
    End If
    c = ParseNum(cboCut.Text)
    If c < 0 Or c > 1000 Then
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("settings_title")
        Exit Sub
    End If
    If cboEnd.ListIndex = 0 Or Len(Trim$(cboEnd.Text)) = 0 Then
        SaveSetting "NeonCenterline", "Params", "EndOff", ""
    ElseIf ParseNum(cboEnd.Text) >= 0 And ParseNum(cboEnd.Text) <= 100 Then
        SaveSetting "NeonCenterline", "Params", "EndOff", FmtNum(ParseNum(cboEnd.Text))
    Else
        MsgBox NeonL("bad_tool"), vbExclamation, NeonL("app_title")
        Exit Sub
    End If
    If Not cboSimp Is Nothing Then
        sm = ParseNum(cboSimp.Text)
        If sm < 0.01 Or sm > 5 Then
            MsgBox NeonL("bad_simp"), vbExclamation, NeonL("settings_title")
            Exit Sub
        End If
        SaveSetting "NeonCenterline", "Simplify", "Tol", FmtNum(sm)
    End If
    SaveSetting "NeonCenterline", "Params", "Tool", FmtNum(t)
    SaveSetting "NeonCenterline", "Cut", "Step", FmtNum(c)
    SaveSetting "NeonCenterline", "Script", "RefEnd", CStr(cboRef.ListIndex)
    SaveSetting "NeonCenterline", "Cut", "Type", IIf(cboType.ListIndex = 1, "neon2", "classic")
    SaveSetting "NeonCenterline", "Cut", "Fit", IIf(chkFit.Value, "1", "0")
    SaveSetting "NeonCenterline", "Cut", "Trim", IIf(cboTrim.ListIndex = 1, "end", "both")
    SaveSetting "NeonCenterline", "UI", "AskTool", IIf(chkAsk.Value, "1", "0")
    SaveSetting "NeonCenterline", "Update", "Daily", IIf(chkUpd.Value, "1", "0")
    Select Case cboLang.ListIndex
    Case 1
        lg = "ru"
    Case 2
        lg = "en"
    Case Else
        lg = ""
    End Select
    SaveSetting "NeonCenterline", "UI", "Lang", lg
    Unload Me
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub

' строка «Упрощение контура, допуск» под выбором режима курсива: всё, что ниже, сдвигается вниз
Private Sub AddSimpRow()
    Dim c As Object, y As Single
    On Error GoTo Fail
    y = cboRef.Top + cboRef.Height + 10
    For Each c In Me.Controls
        If c.Top >= y - 4 Then
            c.Top = c.Top + 26
        End If
    Next c
    Me.Height = Me.Height + 26
    Set lblSimp = Me.Controls.Add("Forms.Label.1", "lblSimp", True)
    lblSimp.Left = 12: lblSimp.Top = y + 4: lblSimp.Width = 170: lblSimp.Height = 18
    Set cboSimp = Me.Controls.Add("Forms.ComboBox.1", "cboSimp", True)
    cboSimp.Left = 190: cboSimp.Top = y: cboSimp.Width = 70: cboSimp.Height = 20
    Exit Sub
Fail:
    Set lblSimp = Nothing
    Set cboSimp = Nothing
End Sub

Private Sub btnUpdate_Click()
    NeonUpdateCheck False
End Sub

Private Sub btnAbout_Click()
    ShowAbout
End Sub
